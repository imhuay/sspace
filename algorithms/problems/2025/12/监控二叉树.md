```python
class Solution:
    def minCameraCover(self, root: Optional[TreeNode]) -> int:
        
        def dfs(n):
            if not n:
                return inf, 0, 0
            
            l_i, l_p, l_c = dfs(n.left)
            r_i, r_p, r_c = dfs(n.right)

            i = min(l_i, l_p, l_c) + min(r_i, r_p, r_c) + 1
            p = min(l_i, l_c) + min(r_i, r_c)
            c = min(l_i + r_c, r_i + l_c, l_i + r_i)
            return i, p, c
        
        i, _, c = dfs(root)
        return min(i, c)
```
