```md
给定两个字符串 text1 和 text2，返回这两个字符串的最长 **公共子序列** 的长度。

class Solution:
    def longestCommonSubsequence(self, s: str, t: str) -> int:
        
        @cache
        def dfs(i, j):
            if i == 0 or j == 0:
                return 0
            
            if s[i - 1] == t[j - 1]:
                r1 = dfs(i - 1, j - 1)
                return r1 + 1
            
            r2 = dfs(i - 1, j)
            r3 = dfs(i, j - 1)
            return max(r2, r3)
        
        m, n = len(s), len(t)
        return dfs(m, n)
```
