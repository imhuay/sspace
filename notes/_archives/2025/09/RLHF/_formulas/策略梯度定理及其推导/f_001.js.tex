    J(\theta)
        &= \underline{\mathbb{E}_{\tau\sim p_\theta(\tau)}} \big\lbrack\, G(\tau) \,\big\rbrack \\
    \nabla_\theta J(\theta)
        &= \underline{\mathbb{E}_{\tau\sim p_\theta(\tau)}} \left\lbrack\ \sum_{t=0}^{\infty} \nabla_\theta\log \pi_\theta(a_t|s_t)\cdot G(\tau) \ \right\rbrack \\
        &= \underline{\mathbb{E}_{\tau\sim p_\theta(\tau)}} \left\lbrack\ \sum_{t=0}^{\infty} \nabla_\theta\log \pi_\theta(a_t|s_t)\cdot \underline{\gamma^t G_t} \ \right\rbrack \\
        &= \sum_{t=0}^{\infty} \underline{\mathbb{E}_{s_t,a_t \sim p_t(s_t,a_t)}} \left\lbrack\ \nabla_\theta\log \pi_\theta(a_t|s_t)\cdot \underline{\gamma^t Q^{\pi_\theta}(s_t, a_t)} \ \right\rbrack
\end{align*}
