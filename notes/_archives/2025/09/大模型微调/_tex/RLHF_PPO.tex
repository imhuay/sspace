\begin{algorithm}
\caption*{\textbf{Algorithm 1:} Step-wise Trajectory Sampling}
\begin{algorithmic}[1]
\Require $\pi_{\theta}, V_{\psi}, R_{\phi}, \pi_{\text{ref}}, \mathcal{D}, \gamma, \beta, \lambda$
\Ensure $\tau$

\State $x \sim \mathcal{D}$ \Comment{Sample prompt}
\State $s_1 \gets x$ \Comment{Init state}
\State $y \gets \text{""}$ \Comment{Init response}
\State $\tau \gets [\ ]$ \Comment{Init trajectory list}
\State $terminal \gets \text{False}$ \Comment{Terminal signal}

\For{$t = 1,2,\dots$}
    % \State $\text{probs}^{\pi_{\theta}}_{t} = \pi_{\theta}(\cdot \mid s_t)$
    % \State $\text{probs}^{\pi_{\theta}}_{t} = \text{Softmax}\big(\text{Forward}(\pi_{\theta}, s_t)\big)$
    \State $\pi_\theta(\cdot|s_t) = \text{Softmax}\big(\text{Forward}(\pi_{\theta}, s_t)\big)$
    % \State $\text{probs}^{\pi_{\text{ref}}}_{t} = \pi_{\text{ref}}(\cdot \mid s_t)$
    % \State $\text{probs}^{\pi_{\theta}}_{t} = \text{Softmax}\big(\text{Forward}(\pi_{\text{ref}}, s_t)\big)$
    \State $\pi_\text{ref}(\cdot|s_t) = \text{Softmax}\big(\text{Forward}(\pi_{\text{ref}}, s_t)\big)$
    % \State $a_t,\, \log p^{a_t} \sim \text{Categorical}(\text{probs}^{\pi_{\theta}}_{t})$ \Comment{Sample action}
    \State $a_t,\, \log\pi_{\theta}(a_t|s_t) \sim \text{Categorical}\big(\pi_\theta(\cdot|s_t)\big)$ \Comment{Sample action}
    \State $s_{t+1} = s_t + a_t$
    \State $y = y + a_t$
    \If{$a_t == [\text{EOS}]$} \Comment{At this time $t{=}{T}$}
        % \State $r_t = -\beta{\cdot}D_{\text{KL}}(\text{probs}^{\pi_{\theta}}_{t}\,\|\,\text{probs}^{\pi_{\text{ref}}}_{t}) + R_\phi(x,y)$
        \State $r_t = -\beta{\cdot}D_{\text{KL}}\big(\pi_\theta(\cdot|s_t) \,\|\, \pi_\text{ref}(\cdot|s_t)\big) + R_\phi(x,y)$
        \State $terminal = \text{True}$
    \Else
        % \State $r_t = -\beta{\cdot}D_{\text{KL}}(\text{probs}^{\pi_{\theta}}_{t}\,\|\,\text{probs}^{\pi_{\text{ref}}}_{t})$
        \State $r_t = -\beta{\cdot}D_{\text{KL}}\big(\pi_\theta(\cdot|s_t) \,\|\, \pi_\text{ref}(\cdot|s_t)\big)$
    \EndIf
    \State Append\big($\tau,\, [s_t, a_t, r_t, \log\pi_{\theta}(a_t|s_t)]$\big) \Comment{Add one}
    \If{$terminal$}
        \State \textbf{break}
    \EndIf
\EndFor

\State $T \gets \text{Length}(\tau)$
\State $\hat{A}_{T+1} \gets 0$ \Comment{Bootstrap final advantage}

\For{$t = T, \dots, 1$}
    \State $(s_t, a_t, r_t, \_) = \tau[t]$
    \State $s_{t+1} = s_t + a_t$
    \State $\delta_t = r_t + \gamma V_{\psi}(s_{t+1}) - V_{\psi}(s_t)$
    \State $\hat{A}_t = \delta_t + \gamma\lambda\,\hat{A}_{t+1}$ \Comment{GAE}
    \State $\hat{R}_t = \hat{A}_t + V_{\psi}(s_t)$ \Comment{Return estimate}
    \State Extend\big($\tau[t],\, [\hat{A}_t, \hat{R}_t]$\big) \Comment{One step $[s_t,a_t,r_t,\log\pi_{\theta}(a_t|s_t),\hat{A}_t,\hat{R}_t]$}
\EndFor

\end{algorithmic}
\end{algorithm}

\begin{algorithm}
\caption*{\textbf{Algorithm 2:} Policy Gradient Update (Mini-Batch)}
\begin{algorithmic}[1]
    \Require $\{(s_i, a_i, \hat{A}_i, \hat{R}_i, \log \pi_{\theta_{\text{old}}}(a_i|s_i))\}_{i=1}^N,\, \pi_{\theta},\, V_{\psi},\, \epsilon,\, c_v,\, c_e,\, \alpha$
    \Ensure $\theta,\ \psi$

    \State $\mathcal{L}^{\text{Total}} \gets 0$ \Comment{Init total loss}

    \For{$i = 1, \cdots, N$}
        % \State $\text{probs}^{\pi_{\theta}}_{s_i} = \text{Softmax}\big(\text{Forward}(\pi_{\theta}, s_i)\big)$ \Comment{Distribution $\pi_\theta(\cdot|s_i)$}
        \State $\pi_\theta(\cdot|s_i) = \text{Softmax}\big(\text{Forward}(\pi_{\theta}, s_i)\big)$ \Comment{New Distribution of $a_i$}
        % \State $\log{\pi_{\theta}}_{\text{new}}^{(a_i|s_i)} \coloneqq \log\text{probs}^{\pi_{\theta}}_{s_i}[a_i]$
        \State $\pi_{\theta}(a_i|s_i) = \text{Gather}(\pi_\theta(\cdot|s_i), a_i)$
        % \State $r_i(\theta) = \exp\!\big(\log\pi_{\theta}(a_i|s_i) - \log{\pi_{\theta}}_{\text{old}}^{(a_i|s_i)}\big)$
        \State $r_i(\theta) = \exp\!\big(\log \pi_{\theta}(a_i|s_i) - \log \pi_{\theta_{\text{old}}}(a_i|s_i)\big)$

        \State $L^{\text{CLIP}}_i = \min\!\big( r_i(\theta)\hat{A}_i,\ \text{clip}(r_i(\theta),1-\epsilon,1+\epsilon)\hat{A}_i \big)$ \Comment{Maximize Policy objective}
        \State $L^{\text{VF}}_i = \tfrac{1}{2}\,\big(V_{\psi}(s_i) - \hat{R}_i\big)^2$ \Comment{Minimize Value loss}
        \State $L^{\text{S}}_i = - \sum_{a} \pi_{\theta}(a|s_i)\log \pi_{\theta}(a|s_i)$ \Comment{Entropy bonus (Optional)}
        % \State $L^{\text{S}}_i = \text{Entropy}(\pi_\theta(\cdot|s_i))$ \Comment{Entropy bonus (Optional)}

        \State $\mathcal{L}^{\text{Total}} \;{+}{=}\; -\big(L^{\text{CLIP}}_i - c_v L^{\text{VF}}_i + c_e L^{\text{S}}_i\big)$ \Comment{Minimize Total loss}
    \EndFor
    
    \State $\mathcal{L}^{\text{Total}} = \tfrac{1}{N}\mathcal{L}^{\text{Total}}$ \Comment{Average over mini-batch}
    \State $(\theta,\psi) \gets (\theta,\psi) - \alpha\nabla_{\theta,\psi}\,\mathcal{L}^{\text{Total}}$
    
\end{algorithmic}
\end{algorithm}
